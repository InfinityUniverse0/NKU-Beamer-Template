\section{背景介绍}

\begin{frame}{研究背景}
  \begin{itemize}
    \item 知识图谱嵌入通过向量表示实体与关系\footfullcite{bordes2013translating}
    \item 思维链提示为大语言模型（LLM）推理提供了一种方法\footfullcite{wei2022chain}
    \item 检索增强生成结合外部知识与语言模型\footfullcite{lewis2020rag}
    \item 本示例关注的研究问题：
    \begin{itemize}
      \item 实体识别与链接的准确率有待提升
      \item 关系抽取在长尾分布下效果不佳
      \item 缺乏统一的端到端构建框架
    \end{itemize}
  \end{itemize}

  \vspace{0.5em}
  \begin{nkremark}[title={}]
    示例主题：结合 LLM 的语义理解能力，探索领域知识图谱构建。
  \end{nkremark}
\end{frame}

\begin{frame}{任务定义与示例约定}
  \begin{nkbox}[title={从文档到知识三元组}]
    \textbf{输入：}文档集合 $\mathcal{D}$ 与可检索知识库 $\mathcal{K}$。\\[0.3em]
    \textbf{输出：}经过验证的三元组集合
    $\widehat{\mathcal{G}} = \{(h,r,t)\}$。\\[0.3em]
    \textbf{评价：}Precision、Recall、F1 与推理时延。
  \end{nkbox}
  \begin{nkremark}[title={如何阅读本示例}]
    后续方法、代码与\alert{合成实验数据}用于演示学术汇报排版，
    不代表真实实验结论。引用的论文用于介绍相关概念。
  \end{nkremark}
\end{frame}

\begin{frame}{基本概念}
  \begin{nkdefinition}[title={知识图谱 (Knowledge Graph)}]
    知识图谱是一种用图结构来建模和存储现实世界中实体及其关系的技术，
    通常表示为三元组集合 $\mathcal{G} = \{(h, r, t)\}$，
    其中 $h, t \in \mathcal{E}$（实体集），$r \in \mathcal{R}$（关系集）。
  \end{nkdefinition}

  \vspace{0.5em}

  \begin{nkbox}[title={TransE 评分函数}]
    TransE 用平移关系 $\bm{h}+\bm{r}\approx\bm{t}$ 建模三元组，
    定义距离评分 $f(h,r,t)=\|\bm{h}+\bm{r}-\bm{t}\|_2$；
    较小的距离表示更高的模型可信度。\footfullcite{bordes2013translating}
  \end{nkbox}
\end{frame}

\begin{frame}{模型框架}
  核心思想：利用 LLM 的上下文学习能力，通过 \textbf{检索增强生成}
  (Retrieval-Augmented Generation)\footfullcite{lewis2020rag} 实现知识三元组的精确抽取。

  \vspace{0.5em}

  目标函数：
  \begin{equation}
    \mathcal{L} = -\sum_{(h,r,t) \in \mathcal{G}}
      \log P_\theta(t \mid h, r, \mathcal{C})
      + \lambda \|\theta\|_2^2
  \end{equation}

  \vspace{0.3em}

  \begin{nkremark}[title={建模说明}]
    $\mathcal{C}$ 为检索上下文，$\lambda\geq 0$ 为正则化系数。
    本目标函数仅作数学排版示例；具体优化性质需结合模型与假设分析。
  \end{nkremark}
\end{frame}

\begin{frame}{检索、生成与验证}

  \end{nkhighlight}
\end{frame}

\begin{frame}{评分函数的稳定性}
  \begin{nklemma}[title={引理：反三角不等式}]
    对任意向量 $\bm{x},\bm{y}$ 及同一种范数，
    $\bigl|\|\bm{x}\|-\|\bm{y}\|\bigr|\leq\|\bm{x}-\bm{y}\|$。
  \end{nklemma}
  \begin{nktheorem}[title={定理：TransE 距离评分的扰动上界}]
    设 $f(\bm{h},\bm{r},\bm{t})=\|\bm{h}+\bm{r}-\bm{t}\|_2$。
    若各嵌入向量的扰动范数均不超过 $\varepsilon$，则
    \[
      |f(\bm{h}+\Delta\bm{h},\bm{r}+\Delta\bm{r},\bm{t}+\Delta\bm{t})
       -f(\bm{h},\bm{r},\bm{t})|\leq 3\varepsilon.
    \]
  \end{nktheorem}
\end{frame}

\begin{frame}{扰动上界的证明}
  \begin{nkproof}[title={证明：两次应用范数不等式}]
    记 $\bm{x}=\bm{h}+\bm{r}-\bm{t}$，
    $\bm{\delta}=\Delta\bm{h}+\Delta\bm{r}-\Delta\bm{t}$。
    由反三角不等式及三角不等式，
    \begin{align*}
      \bigl|\|\bm{x}+\bm{\delta}\|_2-\|\bm{x}\|_2\bigr|
      &\leq\|\bm{\delta}\|_2\\
      &\leq\|\Delta\bm{h}\|_2+\|\Delta\bm{r}\|_2+\|\Delta\bm{t}\|_2\\
      &\leq 3\varepsilon.\qquad\square
    \end{align*}
  \end{nkproof}
  \begin{nkremark}[title={}]
    该界仅说明距离评分对嵌入扰动的稳定性，不等价于抽取准确率保证。
  \end{nkremark}
\end{frame}

\begin{frame}{算法流程}
  \vspace{-0.5em}
  \begin{algorithm}[H]
    \small
    \caption{RAG-based Knowledge Extraction}
    \SetAlgoLined
    \KwIn{Document set $\mathcal{D}$, LLM $\mathcal{M}$, Knowledge base $\mathcal{K}$}
    \KwOut{Knowledge Graph $\mathcal{G}$}
    $\mathcal{G} \gets \emptyset$\;
    \ForEach{$d \in \mathcal{D}$}{
      $\mathcal{C} \gets \text{Retrieve}(\mathcal{K}, d)$;~~%
      $\text{triples} \gets \mathcal{M}(d, \mathcal{C})$\;
      \ForEach{$(h, r, t) \in \text{triples}$}{
        \lIf{$\text{Verify}(h, r, t, \mathcal{K})$}{$\mathcal{G} \gets \mathcal{G} \cup \{(h, r, t)\}$}
      }
    }
    \Return{$\mathcal{G}$}\;
  \end{algorithm}
\end{frame}

\begin{frame}[fragile]{代码示例}
  \begin{lstlisting}[language=Python, caption={Validate and deduplicate triples (Python 3.9+)}]
import json

def parse_triples(text: str) -> list[tuple[str, str, str]]:
    """Parse a JSON array of non-empty string triples."""
    data = json.loads(text)
    if not isinstance(data, list):
        raise ValueError("Expected a JSON array")
    triples = {
        tuple(value.strip() for value in item)
        for item in data
        if isinstance(item, list) and len(item) == 3
        and all(isinstance(v, str) and v.strip() for v in item)
    }
    return sorted(triples)
  \end{lstlisting}
\end{frame}
